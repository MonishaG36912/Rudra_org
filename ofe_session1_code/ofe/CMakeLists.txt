cmake_minimum_required(VERSION 3.20)
project(OrderFlowEngine VERSION 1.0.0 LANGUAGES CXX)

# ── C++ standard ──────────────────────────────────────────────────────────────
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ── Build type defaults ────────────────────────────────────────────────────────
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# ── Compiler flags ────────────────────────────────────────────────────────────
if(MSVC)
    add_compile_options(
        /W4 /WX           # All warnings as errors
        /O2               # Optimise for speed
        /EHsc             # Standard exception handling
        $<$<CONFIG:Debug>:/DDEBUG /Od /Zi>
        $<$<CONFIG:Release>:/DNDEBUG /O2 /GL>  # Whole program optimisation
    )
    add_link_options(
        $<$<CONFIG:Release>:/LTCG /OPT:REF /OPT:ICF>  # Strip unused code
    )
else()
    add_compile_options(
        -Wall -Wextra -Wpedantic -Werror
        -march=native     # Optimise for build machine CPU (unlock AVX/SSE)
        $<$<CONFIG:Debug>:-O0 -g -DDEBUG -fsanitize=address,thread,undefined>
        $<$<CONFIG:Release>:-O3 -DNDEBUG -flto>
    )
    add_link_options(
        $<$<CONFIG:Debug>:-fsanitize=address,thread,undefined>
        $<$<CONFIG:Release>:-flto -Wl,--strip-all>
    )
endif()

# ── Include directories ───────────────────────────────────────────────────────
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

# ── External dependencies ─────────────────────────────────────────────────────
include(FetchContent)

# Google Test — unit testing
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0
)
FetchContent_MakeAvailable(googletest)

# yaml-cpp — configuration file parsing
FetchContent_Declare(
    yaml-cpp
    GIT_REPOSITORY https://github.com/jbeder/yaml-cpp.git
    GIT_TAG        0.8.0
)
FetchContent_MakeAvailable(yaml-cpp)

# ── Source files ──────────────────────────────────────────────────────────────
# (Populated as AGT-03 / AGT-04 / AGT-05 generate implementation files)

set(OFE_CORE_SOURCES
    # AGT-03 outputs (added when generated):
    # src/core/tick_record.cpp
    # src/core/ring_buffer_impl.cpp  (template instantiations)
    # src/core/tick_router.cpp
    # src/core/lee_ready.cpp
    # src/core/bar_types.cpp
    # src/core/bar_engine.cpp
    # src/core/symbol_worker.cpp
    # src/core/event_bus_redis.cpp
)

set(OFE_ANALYTICS_SOURCES
    # AGT-04 outputs (added when generated):
    # src/analytics/delta_engine.cpp
    # src/analytics/imbalance_detector.cpp
    # src/analytics/volume_profile.cpp
    # src/analytics/vwap_engine.cpp
    # src/analytics/signal_detector.cpp
)

set(OFE_FEED_SOURCES
    # AGT-05 outputs (added when generated):
    # src/feed/iqfeed_adapter.cpp
    # src/feed/kinetick_cpp_receiver.cpp
    # src/feed/esignal_adapter.cpp
    # src/feed/feed_manager.cpp
    # src/feed/adapter_factory.cpp
)

set(OFE_LICENSE_SOURCES
    # AGT-09 outputs:
    # src/license/license_engine.cpp
    # src/license/hw_fingerprint.cpp
)

# ── Main engine library ───────────────────────────────────────────────────────
add_library(ofe_engine SHARED
    ${OFE_CORE_SOURCES}
    ${OFE_ANALYTICS_SOURCES}
    ${OFE_FEED_SOURCES}
    ${OFE_LICENSE_SOURCES}
)

target_include_directories(ofe_engine PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

target_link_libraries(ofe_engine PRIVATE
    yaml-cpp
    pthread  # Linux thread support
)

# ── Engine executable ─────────────────────────────────────────────────────────
add_executable(ofe_server
    # src/main.cpp   (added when generated by AGT-03)
)
target_link_libraries(ofe_server PRIVATE ofe_engine)

# ── Unit tests ────────────────────────────────────────────────────────────────
# AGT-10 Test Agent will populate these:
enable_testing()

add_executable(ofe_tests
    # test/test_tick_record.cpp
    # test/test_ring_buffer.cpp
    # test/test_lee_ready.cpp
    # test/test_bar_engine.cpp
    # test/test_delta_engine.cpp
    # test/test_imbalance_detector.cpp
    # test/test_volume_profile.cpp
    # test/test_vwap_engine.cpp
    # test/test_signal_detector.cpp
    # test/test_pulse.cpp
    # test/test_turns.cpp
    # test/test_license_engine.cpp
)
target_link_libraries(ofe_tests PRIVATE
    ofe_engine
    GTest::gtest_main
)
include(GoogleTest)
gtest_discover_tests(ofe_tests)

# ── Synthetic tick generator (stress testing) ─────────────────────────────────
add_executable(ofe_tick_gen
    # test/tick_generator.cpp  (added by AGT-10)
)
target_link_libraries(ofe_tick_gen PRIVATE ofe_engine)

# ── Install ───────────────────────────────────────────────────────────────────
install(TARGETS ofe_engine ofe_server
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)
install(DIRECTORY include/ DESTINATION include/ofe)

# ── Print config summary ──────────────────────────────────────────────────────
message(STATUS "==========================================================")
message(STATUS "  Order Flow Engine v${PROJECT_VERSION}")
message(STATUS "  Build type:    ${CMAKE_BUILD_TYPE}")
message(STATUS "  C++ standard:  ${CMAKE_CXX_STANDARD}")
message(STATUS "  Install prefix: ${CMAKE_INSTALL_PREFIX}")
message(STATUS "==========================================================")
